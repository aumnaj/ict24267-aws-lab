# แบบฝึกหัดที่ 1 — PUT และ DELETE สำหรับ In-Memory list

@app.route('/students/<int:student_id>', methods=['PUT'])
def update_student(student_id):
    data = request.get_json()
    # หา student ใน list ด้วย next()
    student = next((s for s in students if s["id"] == student_id), None)
    if not student:
        return jsonify({"status": "error", "message": "Student not found"}), 404
    # อัปเดตเฉพาะ field ที่ส่งมา
    if "name" in data:
        student["name"] = data["name"]
    if "gpa" in data:
        student["gpa"] = data["gpa"]
    return jsonify({"status": "success", "data": student})

@app.route('/students/<int:student_id>', methods=['DELETE'])
def delete_student(student_id):
    global students
    original_len = len(students)
    students = [s for s in students if s["id"] != student_id]
    if len(students) == original_len:
        return jsonify({"status": "error", "message": "Student not found"}), 404
    return jsonify({"status": "success", "message": f"Student {student_id} deleted"})

จุดสำคัญที่ต้องรู้: ใช้ next() หาข้อมูลใน list, กำหนด HTTP status code ที่ถูกต้อง (404 เมื่อไม่พบ, 200 เมื่อสำเร็จ), ใช้ global เมื่อต้องการแก้ไข list ระดับ module

Security Groups: Inbound vs Outbound อธิบายความแตกต่าง พร้อมยกตัวอย่างว่าทำไม Outbound ก็สำคัญ
▾
✅ เฉลย
  • Inbound rules: ควบคุม traffic ที่เข้ามา จาก Internet → Instance เช่น อนุญาต Port 80 (HTTP), 22 (SSH)
  • Outbound rules: ควบคุม traffic ที่ออกไป จาก Instance → Internet/Services อื่น

ทำไม Outbound ก็สำคัญ?

  • ถ้า EC2 ถูก hack แล้วมี malware ส่ง data ออก — Outbound ที่ restrictive จะบล็อกได้
  • ในสภาพแวดล้อม Production ควรจำกัด Outbound ให้เชื่อมต่อได้เฉพาะ services ที่จำเป็น เช่น RDS (3306), S3, CloudWatch เท่านั้น
  • AWS Security Group เป็น Stateful — connection ที่ Inbound อนุญาตแล้ว response จะออกได้โดยอัตโนมัติ แม้ Outbound จะ block

Flask Architecture ทำไมถึงต้องใช้ Nginx เป็น Reverse Proxy หน้า Flask แทนที่จะให้ Flask รับ request โดยตรง?
▾
✅ เฉลย

Flask ไม่เหมาะกับ Production โดยตรง เพราะ:

  • Single-threaded: Flask development server รับ request ได้ทีละ 1 ใน default mode
  • Port 5000: ต้องระบุ port ในทุก URL, ไม่เป็น standard
  • ไม่มี security features: ไม่มี DDoS protection, rate limiting, SSL termination

ประโยชน์ของ Nginx Reverse Proxy:

  • รับ request ทาง Port 80 (HTTP standard) — ไม่ต้องใส่ :5000 ใน URL
  • Static file serving ที่เร็วกว่า — Nginx เก่งกว่า Flask มาก
  • Load balancing — กระจาย request ไปหลาย Flask process ได้
  • SSL Termination — จัดการ HTTPS ที่ Nginx แทนที่จะให้ Flask ทำ
  • Buffer requests — ป้องกัน slow clients ไม่ให้กิน Flask resource

@app.route('/enrollments', methods=['POST'])
def create_enrollment():
    try:
        data = request.get_json()
        new_id = execute_query(
            "INSERT INTO enrollments (student_id, course_id, semester) VALUES (%s, %s, %s)",
            (data['student_id'], data['course_id'], data.get('semester', '2/2568')),
            fetch=False
        )
        return jsonify({"status": "success", "enrollment_id": new_id}), 201
    except Exception as e:
        return jsonify({"status": "error", "message": str(e)}), 500

# 2. GET /students/<id>/courses — วิชาที่นักศึกษาลงทะเบียน
@app.route('/students/<int:sid>/courses', methods=['GET'])
def student_courses(sid):
    result = execute_query(
        """SELECT c.course_code, c.course_name, e.grade, e.semester
           FROM enrollments e JOIN courses c ON e.course_id = c.id
           WHERE e.student_id = %s""", (sid,)
    )
    return jsonify({"status": "success", "data": result})

# 3. GET /courses/<id>/students — นักศึกษาในวิชา
@app.route('/courses/<int:cid>/students', methods=['GET'])
def course_students(cid):
    result = execute_query(
        """SELECT s.student_id, s.name, e.grade, e.semester
           FROM enrollments e JOIN students s ON e.student_id = s.id
           WHERE e.course_id = %s""", (cid,)
    )
    return jsonify({"status": "success", "data": result})

จุดสำคัญ: ใช้ JOIN เชื่อมตาราง, ส่ง fetch=False สำหรับ INSERT, ระบุ HTTP status 201 Created สำหรับสร้างข้อมูลใหม่

RDS Snapshot

ขั้นตอน: RDS → Databases → ict24267-db → Actions → Take snapshot → ตั้งชื่อ ict24267-db-week2-backup → Take Snapshot

ความแตกต่างของ Backup ใน RDS 2 แบบ:

  • Automated Backup: รันทุกวันอัตโนมัติ, เก็บได้ 1-35 วัน (ตั้งใน Retention period), ลบอัตโนมัติเมื่อ DB หมดอายุ

2
Security Best Practiceทำไม RDS ต้องตั้ง Public access: No? ความเสี่ยงถ้าเปิด Public?
▾
✅ เฉลย

เหตุผลที่ต้อง Public access: No:

  • Database ควรอยู่ใน Private Subnet ที่ไม่สามารถเข้าถึงจาก Internet ได้โดยตรง
  • ให้ EC2 เป็น "Jump Server" หรือ gateway — traffic ผ่าน Application layer ก่อน
  • ลด Attack Surface — bot scanner จาก Internet ไม่เห็น DB endpoint

ความเสี่ยงถ้าเปิด Public access: Yes:

  • Brute Force Attack: bot พยายาม login ด้วย password ต่างๆ ตลอด 24 ชม.
  • SQL Injection จาก Internet: ถ้า password รั่วออกไป ใครก็เข้า DB ได้โดยตรง
  • Data Exfiltration: dump ข้อมูลออกได้ทันทีโดยไม่ผ่าน Application
  • Zero-day Exploit: MySQL vulnerability ถูกโจมตีได้โดยตรง

Connection Poolingถ้ามีผู้ใช้ 1,000 คน สร้าง DB connection ใหม่ทุก request — ปัญหาคืออะไร? Pool แก้อย่างไร?
▾
✅ เฉลย

ปัญหาที่เกิดขึ้น:

  • Connection Overhead: การสร้าง TCP connection + MySQL handshake ใช้เวลา 10-100ms ทุกครั้ง — ช้ามาก
  • MySQL Connection Limit: MySQL มี max_connections (default ~150) ถ้า 1,000 request พร้อมกัน จะ error "Too many connections"
  • Resource Waste: แต่ละ connection กิน RAM ของ MySQL server

Connection Pool แก้อย่างไร:

  • สร้าง connection ไว้ล่วงหน้าจำนวนหนึ่ง (เช่น 10-20 connections)
  • แต่ละ request "ขอยืม" connection จาก pool — เมื่อเสร็จแล้ว "คืน" ไม่ได้ปิด
  • ลด overhead ต่อ request จาก ~50ms เหลือ <1ms สำหรับการขอ connection
  • ใน Python ใช้ SQLAlchemy หรือ DBUtils.PooledDB

SQL Injectionทำไม Parameterized Query (%s) ถึงปลอดภัยกว่า String Formatting?
▾
✅ เฉลย

String Formatting (อันตราย ❌):

ห้ามทำแบบนี้
# ❌ อันตราย — SQL Injection ได้
name = request.args.get('name')  # ถ้า user ใส่: "'; DROP TABLE students; --"
query = f"SELECT * FROM students WHERE name = '{name}'"
# จะกลายเป็น: SELECT * FROM students WHERE name = ''; DROP TABLE students; --'

Parameterized Query (ปลอดภัย ✅):

วิธีที่ถูกต้อง
# ✅ ปลอดภัย — PyMySQL จัดการ escape ให้
cursor.execute("SELECT * FROM students WHERE name = %s", (name,))
# PyMySQL จะ escape ทุกตัวอักษรพิเศษอัตโนมัติ
# "'; DROP TABLE students; --" จะถูกมองเป็น string ธรรมดา ไม่ใช่ SQL command

หลักการ: Parameterized Query แยก "โครงสร้าง SQL" ออกจาก "ค่าข้อมูล" — database driver จัดการ escaping ให้ก่อน query ถูกส่งไปยัง MySQL ทำให้ user input ไม่มีทางเป็น SQL command ได้

DynamoDB vs RDSเปรียบเทียบใน 4 ด้าน: โครงสร้างข้อมูล, query, ค่าใช้จ่าย, ความเหมาะสม
▾
✅ เฉลย
  • โครงสร้างข้อมูล: DynamoDB = flexible (NoSQL, JSON-like, ไม่มี schema), RDS = fixed schema (ตาราง, columns, data types ต้องกำหนดล่วงหน้า)
  • การ Query: DynamoDB = จำกัด (ต้องรู้ Partition Key, ไม่มี JOIN), RDS = ยืดหยุ่นมาก (JOIN หลายตาราง, aggregate functions, subquery)

  • ความเหมาะสม: DynamoDB → CRUD ง่าย, high traffic, gaming leaderboard, session storage; RDS → รายงาน, financial data, ข้อมูลที่มี relation ซับซ้อน

Cold Startเกิดจากอะไร? มีผลกระทบอย่างไรต่อ User Experience?
▾
✅ เฉลย

Cold Start เกิดจาก: เมื่อไม่มี request มาสักพัก Lambda จะ "หลับ" (terminate container) — ครั้งถัดไปที่มี request AWS ต้องทำ:

  • Download code จาก S3 (~50-100ms)
  • สร้าง container ใหม่ (~100-200ms)
  • Initialize Python runtime + import libraries (~100-500ms)
  • รัน code จริง

รวมกันอาจใช้เวลา 500ms-3 วินาที — Warm start (container ยังอยู่) ใช้แค่ <100ms

ผลกระทบต่อ UX: User รู้สึก "ช้า" ครั้งแรก โดยเฉพาะ API ที่ไม่ค่อยมีคนใช้

วิธีลด Cold Start:

  • Provisioned Concurrency: ให้ Lambda "warm" ตลอดเวลา (มีค่าใช้จ่าย)
  • Warm-up ping: ส่ง request ปลอมทุก 5 นาทีเพื่อป้องกัน Lambda หลับ
  • ลด package size: import เฉพาะที่ต้องใช้, ใช้ Lambda Layer

IAM Role & Least PrivilegeFunction จริงๆ ต้องการ permission อะไรบ้าง? การให้เกินมีความเสี่ยงอะไร?
▾
✅ เฉลย

Permission ที่จำเป็นจริงๆ สำหรับ submit-feedback Lambda:

  • dynamodb:PutItem — เฉพาะ table ict24267-feedbacks เท่านั้น
  • logs:CreateLogGroup, logs:CreateLogStream, logs:PutLogEvents — สำหรับ CloudWatch Logs
  • sns:Publish — เฉพาะ topic ARN ที่ระบุ

แต่เราให้ AmazonDynamoDBFullAccess ซึ่งรวมถึง DeleteTable, CreateTable, Scan ทุก table

ความเสี่ยงของการให้ permission เกิน (Over-privileged):

  • ถ้า code มี bug → อาจ delete table ที่ไม่ต้องการโดยไม่ตั้งใจ
  • ถ้า Lambda ถูก compromise → attacker ได้ access เต็มๆ ของ DynamoDB ทั้งบัญชี
  • ละเมิดหลัก Principle of Least Privilege (ให้สิทธิ์น้อยที่สุดเท่าที่จำเป็น)

เลือก Lambda + DynamoDB เมื่อ:

  • Traffic ไม่แน่นอน หรือกระจุกตัวบางช่วงเวลา
  • ต้องการลดค่าใช้จ่าย (จ่ายตามใช้จริง)
  • ไม่อยากดูแล Server — ทีมเล็ก focus ที่ business logic
  • ข้อมูลไม่มี relation ซับซ้อน (CRUD ง่ายๆ)

เลือก EC2 + RDS เมื่อ:

  • ต้องการ SQL query ซับซ้อน (JOIN หลายตาราง, aggregate)
  • มี background process รันตลอด (WebSocket, queue consumer)
  • ระบบเดิมเป็น traditional architecture — migrate ยาก
  • ต้องการ ACID compliance ที่แน่นอน
